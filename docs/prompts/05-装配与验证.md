## 8. 错误与边界

### 8.1 F9：`includePiDefaultGuidelines` 的 pi-rp bug（**前置依赖**）

- **根因**：渲染器读 `ctx.item.options?.includePiDefaultGuidelines !== false`（默认 on、可显式关，`slot-renderers.ts:132`），而 normalizer 只保留 `=== true`（`loader.ts:349`），`false` 被丢；`knownKeys`（`:375`）又挡住未知键透传。相邻的 `onlyWithSnippets` 是同语义却写对了（`=== false`），故判定**疏漏而非设计**。
- **实测**：修前 `false`/`true` 输出**逐字相同**（`ab-false.json` / `ab-true.json` 均含 `Be concise in your responses`，各 114 行）；修后 `false` 剥掉 3 条 Pi guideline（`afx-false.json` 75 行不含、`afx-true.json` 78 行含）。
- **处置**：已在本仓 `vendor/pi-rp` 副本补掉（`src/core/prompt-preset/loader.ts` 一处 6 行），`pnpm pi build` 重建 dist，A/B 复验通过。子模块改动**只有这一处**（`git -C vendor/pi-rp diff --stat` 已确认）。
- **后果（写进 §10 落地顺序）**：**这条修复只在本机 dist 里**。别人 clone 后必须 `pnpm pi commit`（同步子模块指针）+ `pnpm pi build`（重建 dist），否则 `probe-prompt.mjs` 的 `Be concise` 断言在别人机器上会**莫名红**。**`pnpm pi build` 必须先于任何"提示词接线正确"的验证。**

>

**本批实测的 15 个 snippet 文本**（供评审核对；由上面那条 `node -e` 命令产生）：

```

 * `extensions/` and `<worldRoot>/extensions/`, so this file can never be loaded
 * by a production run. The probe passes it explicitly with `--extension`.
 *
 * `RpcClient` runs the agent as a CHILD process, so the probe cannot read this
 * process's memory — the dump FILE is the only observation channel (same reason
 * as `inject-probe-provider.ts:8-11`).
 *
 * It records exactly the two things docs/prompts/00 §2 F1/F4 define as the wire:
 *   `context.systemPrompt` (always "") and `context.messages` (the real system
 *   text is `messages[0]`), plus each tool's `name`/`description` on
 *   `context.tools`. It scripts ONE plain-text turn — the probe only needs the
 *   first request, not a tool loop.
 */
import fs from 'node:fs';
import { createAssistantMessageEventStream } from '@earendil-works/pi-ai/compat';

const PROVIDER_ID = 'airp-prompt-dump';
const MODEL_ID = 'deterministic';

/** Flatten a wire message's content (string | block[]) to plain text. */
function textOf(content: unknown): string {
  if (typeof content === 'string') return content;
  if (!Array.isArray(content)) return '';
  return content
    .map((block: any) => (block && typeof block.text === 'string' ? block.text : ''))
    .filter((s: string) => s !== '')
    .join('\n');
}

function usage(): Record<string, unknown> {
  return {
    input: 10,
    output: 20,
    cacheRead: 0,
    cacheWrite: 0,
    totalTokens: 30,
    cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0, total: 0 },
  };
}

function assistantMessage(model: any, content: unknown[], stopReason: string): Record<string, unknown> {
  return {
    role: 'assistant',
    content,
    api: model.api,
    provider: model.provider,
    model: model.id,
    usage: usage(),
    stopReason,
    timestamp: Date.now(),
  };
}

/** Streams one complete assistant message as start → per-block deltas → done. */
function streamScript(stream: any, message: any): void {
  stream.push({ type: 'start', partial: { ...message, content: [], stopReason: 'pending' } });
  message.content.forEach((block: any, i: number) => {
    if (block.type !== 'text') return;
    stream.push({ type: 'text_start', contentIndex: i, partial: message });
    stream.push({ type: 'text_delta', contentIndex: i, delta: block.text, partial: message });
    stream.push({ type: 'text_end', contentIndex: i, content: block.text, partial: message });
  });
  stream.push({ type: 'done', reason: message.stopReason, message });
  stream.end(message);
}

export default function promptDumpProvider(pi: any): void {
  pi.registerProvider(PROVIDER_ID, {
    name: 'AIRP Prompt Dump (deterministic)',
    baseUrl: 'http://127.0.0.1:0',
    apiKey: 'probe-key',
    api: 'openai-completions',
    models: [
      {
        id: MODEL_ID,
        name: 'Deterministic prompt-dump model',
        reasoning: false,
        input: ['text'],
        cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0 },
        contextWindow: 128000,
        maxTokens: 4096,
      },
    ],
    streamSimple: (model: any, context: any) => {
      const out = process.env.AIRP_PROMPT_OUT;
      if (out) {
        try {
          fs.writeFileSync(
            out,
            JSON.stringify(
              {
                // "" in preset mode — the real system content is messages[0] (F1).
                systemPrompt: context?.systemPrompt ?? null,
                messages: (context?.messages ?? []).map((m: any) => ({
                  role: m.role,
                  text: textOf(m.content),
                })),
                tools: (context?.tools ?? []).map((t: any) => ({ name: t.name, description: t.description })),
              },
              null,
              2
            )
          );
        } catch {
          // The probe fails its "no dump written" assertion; never derail the run.
        }
      }
      const stream = createAssistantMessageEventStream();
      const message = assistantMessage(model, [{ type: 'text', text: 'prompt dump turn done' }], 'stop');
      queueMicrotask(() => streamScript(stream, message));
      return stream;
    },
  });
}
